LiCriterion.summable_li_symmetrized : ∀ (n : ℕ),
  Summable fun ρ =>
    ↑(analyticOrderNatAt LiCriterion.riemannXi ↑ρ) * (1 - (1 - 1 / ↑ρ) ^ (-(↑n + 1)) + (1 - (1 - 1 / ↑ρ) ^ (↑n + 1)))
LiCriterion.taylorCoeff_eq_li_symmetrized : ∀ (n : ℕ),
  LiCriterion.taylorCoeff LiCriterion.riemannXi n =
    2⁻¹ *
      ∑' (ρ : LiCriterion.NontrivialZero),
        ↑(analyticOrderNatAt LiCriterion.riemannXi ↑ρ) *
          (1 - (1 - 1 / ↑ρ) ^ (-(↑n + 1)) + (1 - (1 - 1 / ↑ρ) ^ (↑n + 1)))
