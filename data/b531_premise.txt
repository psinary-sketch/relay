### def is_xi_zero  (Kernel/XiDef.lean ; 5 lines ; identical at HEAD True)
def is_xi_zero (sigma : Real) : Prop :=
  Exists (fun t : Real =>
    riemannZeta (⟨sigma, t⟩ : ℂ) = 0 /\
    (Not (Exists (fun n : Nat => (⟨sigma, t⟩ : ℂ) = -2 * (↑n + 1)))) /\
    (⟨sigma, t⟩ : ℂ) ≠ 1)
