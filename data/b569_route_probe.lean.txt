def modOf (env : Environment) (c : Name) : Name :=
  match env.getModuleIdxFor? c with
  | some i => env.header.moduleNames[i.toNat]!
  | none => Name.anonymous

def walk (env : Environment) (start : Name) : Array (Name × Name) := Id.run do
  let some ci := env.find? start | return #[]
  let mut stack : Array (Name × Name) := ci.getUsedConstantsAsSet.toArray.map (fun d => (d, start))
  let mut seen : NameSet := ({} : NameSet).insert start
  let mut out : Array (Name × Name) := #[]
  while stack.size > 0 do
    let (c, p) := stack.back!
    stack := stack.pop
    if seen.contains c then continue
    seen := seen.insert c
    if (modOf env c).getRoot != `Zeta23 then continue
    out := out.push (c, p)
    if let some cj := env.find? c then
      for d in cj.getUsedConstantsAsSet.toArray do
        unless seen.contains d do stack := stack.push (d, c)
  return out

end RouteProbe
