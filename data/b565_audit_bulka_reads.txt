theorem li_criterion : RiemannHypothesis ↔ ∀ (n : ℕ), 0 ≤ (LiChallenge.taylorCoeff LiChallenge.riemannXi n).re :=
LiCriterion.li_criterion_rh_iff
theorem LiCriterion.li_criterion_rh_iff : RiemannHypothesis ↔
  ∀ (n : ℕ), 0 ≤ (LiCriterion.taylorCoeff LiCriterion.riemannXi n).re :=
LiCriterion.biconditional_rh_li_of_hadamard_order_one LiCriterion.xi_hasFiniteOrder LiCriterion.xi_order_le_one
LiCriterion.li_pos_implies_rh : (∀ (n : ℕ), 0 ≤ (LiCriterion.taylorCoeff LiCriterion.riemannXi n).re) →
  ∀ (s : ℂ), riemannZeta s = 0 → 0 < s.re ∧ s.re < 1 → s.re = 1 / 2
def LiCriterion.taylorCoeff : (ℂ → ℂ) → ℕ → ℂ :=
fun f n => deriv^[n] (LiCriterion.logDeriv (LiCriterion.phi f)) 0 / ↑n.factorial
def LiCriterion.liPairedSummand : ℕ → LiCriterion.NontrivialZero → ℂ :=
fun n ρ => LiCriterion.liSummand n ρ + LiCriterion.liSummand n (LiCriterion.pairedZero ρ)
