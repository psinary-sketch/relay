li_coefficients_eq_zero_sum : ∀ (n : ℕ),
  (Summable fun ρ =>
      ↑(analyticOrderNatAt LiChallenge.riemannXi ↑ρ) *
        (1 - (1 - 1 / ↑ρ) ^ (-(↑n + 1)) + (1 - (1 - 1 / ↑ρ) ^ (↑n + 1)))) ∧
    LiChallenge.taylorCoeff LiChallenge.riemannXi n =
      2⁻¹ *

def LiChallenge.riemannXi : ℂ → ℂ :=
fun s => 1 / 2 * s * (s - 1) * completedRiemannZeta₀ s + 1 / 2
def LiChallenge.taylorCoeff : (ℂ → ℂ) → ℕ → ℂ :=
fun f n => deriv^[n] (LiChallenge.logDeriv (LiChallenge.phi f)) 0 / ↑n.factorial
def LiChallenge.phi : (ℂ → ℂ) → ℂ → ℂ :=
fun f z => f (1 / (1 - z))
def LiChallenge.logDeriv : (ℂ → ℂ) → ℂ → ℂ :=
fun φ z => deriv φ z / φ z
def LiChallenge.NontrivialZero : Type :=
{ ρ // riemannZeta ρ = 0 ∧ 0 < ρ.re ∧ ρ.re < 1 }
def LiCriterion.riemannXi : ℂ → ℂ :=
fun s => 1 / 2 * s * (s - 1) * completedRiemannZeta₀ s + 1 / 2
def LiCriterion.NontrivialZero : Type :=
{ ρ // riemannZeta ρ = 0 ∧ 0 < ρ.re ∧ ρ.re < 1 }
def RiemannHypothesis : Prop :=
∀ (s : ℂ), riemannZeta s = 0 → (¬∃ n, s = -2 * (↑n + 1)) → s ≠ 1 → s.re = 1 / 2
