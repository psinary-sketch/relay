def block(start_prefix, end_prefix):
    a = next(i for i, l in enumerate(src) if l.startswith(start_prefix))
    b = next(i for i, l in enumerate(src) if i > a and l.startswith(end_prefix))
    return '\n'.join(src[a:b]).rstrip() + '\n'
