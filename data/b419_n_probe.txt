  | theorem pow_mod_self (r j : Nat) (hr : 0 < r) (hj : 1 ≤ j) : (r ^ j) % r = 0 := by

  |   exact mul_mod_self' r _ hr

  | theorem mod_r_one (r u j : Nat) (hr2 : 2 ≤ r) (hj : 1 ≤ j) (h : u % r ^ j = 1) : u % r = 1 := by
  |   have hr0 : 0 < r := Nat.lt_of_lt_of_le (by decide) hr2

  |   rw [e, ← B329.mul_assoc'] at hc

  | theorem mod_one_not_dvd (r u j : Nat) (hr2 : 2 ≤ r) (hj : 1 ≤ j) (h : u % r = 1) : u % r ^ j ≠ 0 := by
  |   have hr0 : 0 < r := Nat.lt_of_lt_of_le (by decide) hr2
  |   intro hz
  |   have e := mod_trans (pow_pos' r hr0 j) hr0 hz (pow_mod_self r j hr0 hj)
  |   rw [h] at e

  | theorem cong_of (r s' k L u : Nat) (hr : isPrime r = true) (hs' : s' % r ≠ 0) (hkL : k < L) (hu : 0 < u)
  |     (h : ((u - 1) * (s' * r ^ k)) % (r ^ L) = 0) : u % (r ^ (L - k)) = 1 := by
  |   have hr2 := isPrime_two_le hr
  |   have hr0 : 0 < r := Nat.lt_of_lt_of_le (by decide) hr2

  |   have h2 := pow_cancel r ((u - 1) * s') k L hr0 (Nat.le_of_lt hkL) h

  | theorem cong_to (r s' k L u : Nat) (hr0 : 0 < r) (hkL : k ≤ L) (h : u % (r ^ (L - k)) = 1) :
  |     ((u - 1) * (s' * r ^ k)) % (r ^ L) = 0 := by
  |   obtain ⟨c, hc, _⟩ := mod_spec u (r ^ (L - k)) (pow_pos' r hr0 _)
  |   rw [h] at hc
  |   have e1 : u - 1 = r ^ (L - k) * c := by rw [hc]; all_goals rfl
  |   have e2 : (u - 1) * (s' * r ^ k) = r ^ L * (c * s') := by
  |     rw [e1]
  |     calc r ^ (L - k) * c * (s' * r ^ k) = r ^ (L - k) * (c * (s' * r ^ k)) :=
  |           (B329.mul_assoc' (r ^ (L - k)) c (s' * r ^ k)).symm
  |       _ = r ^ (L - k) * (c * s' * r ^ k) := congrArg (fun x => r ^ (L - k) * x) (B329.mul_assoc' c s' (r ^ k))
  |       _ = r ^ (L - k) * (r ^ k * (c * s')) := congrArg (fun x => r ^ (L - k) * x) (Nat.mul_comm (c * s') (r ^ k))
  |       _ = r ^ (L - k) * r ^ k * (c * s') := B329.mul_assoc' (r ^ (L - k)) (r ^ k) (c * s')
  |       _ = r ^ L * (c * s') := by rw [← B329.pow_add', sub_add_cancel' hkL]

  |   exact mul_mod_self' _ _ (pow_pos' r hr0 L)

  | theorem point (r e E L k s' p q N m s u : Nat) (hr : isPrime r = true) (he : 1 ≤ e) (hs' : s' % r ≠ 0)
  |     (hk : k < E) (hEL : E ≤ L) (hp : p = r ^ e) (hq : q = r ^ E) (hN : N = q * q) (hm : m = r ^ L)
  |     (hs : s = s' * r ^ k) (hoff : s % q ≠ 0) :

  |     have c2 : (!(s % q == 0)) = true := by rw [ne_beq_false hoff]; all_goals rfl
  |     have hq0 : 0 < q := by rw [hq]; exact pow_pos' r hr0 E
  |     have hus : (u * s) % q ≠ 0 := by rw [hs, hq]; exact offball_of r s' k E u hr hs' hk hur

  |       rw [hN, mod_mod_of_mul (u * s) q q hq0 hq0, ne_beq_false hus]; all_goals rfl
  |     have c4 : (((u - 1) * s) % m == 0) = true := by
  |       rw [hs, hm, cong_to r s' k L u hr0 (Nat.le_of_lt hkL) h1]; all_goals rfl

  |       rw [h5] at hb

  |     show p ^ n = r ^ (e * n)
  |     rw [hp, pow_mul']
  |   have hNr : B329.gridN p n = r ^ (e * n + e * n) := by
  |     show p ^ (2 * n) = r ^ (e * n + e * n)
  |     rw [hp, pow_mul', Nat.two_mul, Nat.mul_add]
  |   have hNqq : B329.gridN p n = B329.ballQ p n * B329.ballQ p n := by rw [hNr, hq, B329.pow_add']

  |   apply sumf_congr
  |   intro s _
  |   dsimp only

  |     obtain ⟨k, s', hs, hns', hk⟩ := offball_val r e n s hr2 (by rw [← hq]; exact hoff)
